2 · Perintah setup

npx wrangler login
npx wrangler d1 create rag
npx wrangler r2 bucket create rag-docs
npx wrangler vectorize create rag-index --dimensions=1024 --metric=cosine
npx wrangler vectorize create-metadata-index rag-index \
  --property-name=source --type=string
npx wrangler vectorize create-metadata-index rag-index \
  --property-name=created --type=number

npx wrangler d1 execute rag --remote --file=schema.sql
npx wrangler deploy

Metadata index wajib dibuat sebelum vektor pertama di-upsert. Dimensi dan metrik terkunci setelah indeks dibuat.

CREATE INDEX IF NOT EXISTS idx_chunks_source ON chunks(source);

5 · Chunking

function chunkText(text: string, size = 1200, overlap = 200): string[] {
  const paragraphs = text.split(/\n{2,}/).map((p) => p.trim()).filter(Boolean);
  const chunks: string[] = [];
  let buf = "";

  for (const p of paragraphs) {
    if (buf.length > 0 && buf.length + p.length + 2 > size) {
      chunks.push(buf.trim());
      buf = buf.slice(Math.max(0, buf.length - overlap)).trim() + "\n\n";
    }
    buf += p + "\n\n";
  }
  if (buf.trim()) chunks.push(buf.trim());
  return chunks;
}

7 · Ingestion (Workflow)

import { WorkflowEntrypoint } from "cloudflare:workers";
import type { WorkflowEvent, WorkflowStep } from "cloudflare:workers";

export class IngestWorkflow extends WorkflowEntrypoint<Env, { key: string }> {
  async run(event: WorkflowEvent<{ key: string }>, step: WorkflowStep) {
    const { key } = event.payload;

    const chunks = await step.do("extract + chunk", async () => {
      const text = await extract(this.env, key);
      return chunkText(text);
    });

    for (let i = 0; i < chunks.length; i += 20) {
      await step.do(`embed ${i}`, async () => {
        const batch = chunks.slice(i, i + 20);
        const { data } = await this.env.AI.run("@cf/baai/bge-m3", { text: batch });

        await this.env.INDEX.upsert(
          batch.map((_, j) => ({
            id: `${key}#${i + j}`,
            values: data[j],
            metadata: { source: key, chunk: i + j },
          })),
        );

        await this.env.DB.batch(
          batch.map((text, j) =>
            this.env.DB.prepare(
              `INSERT OR REPLACE INTO chunks
               (id, source, chunk_index, text, hash, created_at)
               VALUES (?, ?, ?, ?, ?, ?)`,
            ).bind(`${key}#${i + j}`, key, i + j, text, "", Date.now()),
          ),
        );
      });
    }
  }
}

Tambahkan hash isi untuk melewati chunk yang tidak berubah. Batch 20 menjaga tiap langkah tetap pendek dan bisa di-retry.

8 · Query

const { data } = await env.AI.run("@cf/baai/bge-m3", { text: [question] });

const { matches } = await env.INDEX.query(data[0], {
  topK: 20,
  returnMetadata: "all",
  filter: { source: { $in: ["catatan/rag.md"] } },
});

Filter diterapkan sebelum top-K. topK maksimum 50 bila metadata diminta, 100 bila tidak.

9 · Ambil teks, jaga urutan

const ids = matches.map((m) => m.id);
const placeholders = ids.map(() => "?").join(",");

const { results } = await env.DB.prepare(
  `SELECT id, source, chunk_index, text
   FROM chunks WHERE id IN (${placeholders})`,
).bind(...ids).all();

const byId = new Map(results.map((r) => [r.id, r]));
const candidates = matches
  .map((m) => ({ ...byId.get(m.id), score: m.score }))
  .filter((c) => c.text);

10 · Rerank

const { response } = await env.AI.run("@cf/baai/bge-reranker-base", {
  query: question,
  contexts: candidates.map((c) => ({ text: c.text })),
  top_k: 6,
});

const top = response
  .map((r) => ({ ...candidates[r.id], rerankScore: r.score }))
  .sort((a, b) => b.rerankScore - a.rerankScore);

11 · Fusion (hybrid)

function rrf(lists: string[][], k = 60): [string, number][] {
  const scores = new Map<string, number>();
  for (const list of lists) {
    list.forEach((id, i) => {
      scores.set(id, (scores.get(id) ?? 0) + 1 / (k + i + 1));
    });
  }
  return [...scores.entries()].sort((a, b) => b[1] - a[1]);
}

15 · Hapus satu sumber

const { results } = await env.DB
  .prepare("SELECT id FROM chunks WHERE source = ?")
  .bind(source)
  .all();

const ids = results.map((r) => r.id);

await env.DB.prepare("DELETE FROM chunks WHERE source = ?").bind(source).run();
if (ids.length) await env.INDEX.deleteByIds(ids);
await env.DOCS.delete(source);

Delete Vectorize asinkron: butuh beberapa detik sampai hilang dari hasil query.
