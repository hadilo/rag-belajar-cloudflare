npx wrangler vectorize create-metadata-index rag-index \
  --property-name=source --type=string

npx wrangler d1 execute rag --remote --file=schema.sql

Metadata index wajib dibuat sebelum vektor pertama di-upsert. Dimensi dan metrik terkunci setelah indeks dibuat.

4 · Embedding

const vectors = await embed(env, ["satu string", "atau array string"]);
// vectors: number[][], masing-masing 1024 angka
// urutan hasil sama dengan urutan input

Wrapper di src/ai.ts mengisolasi cast tipe, karena tipe hasil wrangler types untuk model ini tidak akurat (input reranker kehilangan query, output embedding berupa union). Jangan memanggil env.AI.run langsung dari modul domain.

type AnyRun = (model: string, input: unknown, options?: unknown) => Promise<unknown>;

function run(env: Env): AnyRun {
	// `.bind` penting: env.AI.run bergantung pada `this`.
	return env.AI.run.bind(env.AI) as unknown as AnyRun;
}

export async function embed(env: Env, text: string[]): Promise<number[][]> {
	const out = (await run(env)("@cf/baai/bge-m3", { text })) as { data?: number[][] };
	return out.data ?? [];
}

5 · Chunking

function chunkText(text: string, size = 1200, overlap = 200): string[] {
  const paragraphs = text.split(/\n{2,}/).map((p) => p.trim()).filter(Boolean);
  const chunks: string[] = [];
  let buf = "";

  for (const p of paragraphs) {
    if (buf.length > 0 && buf.length + p.length + 2 > size) {
      chunks.push(buf.trim());
      buf = buf.slice(Math.max(0, buf.length - overlap)).trim() + "\n\n";
    }
    buf += p + "\n\n";
  }
  if (buf.trim()) chunks.push(buf.trim());
  return chunks;
}

6 · Ekstraksi

export class ObjectNotFoundError extends Error {
	constructor(key: string) {
		super(`Tidak ada objek: ${key}`);
		this.name = "ObjectNotFoundError";
	}
}

export async function extract(env: Env, key: string) {
	const obj = await env.DOCS.get(key);
	if (!obj) throw new ObjectNotFoundError(key);

	if (/\.(md|txt|markdown)$/i.test(key)) return await obj.text();

	const [result] = await env.AI.toMarkdown([
		{ name: key, blob: new Blob([await obj.arrayBuffer()]) },
	]);

	if (!result) throw new Error(`Konversi gagal: ${key}`);
	if (result.format === "error") throw new Error(result.error);
	return result.data;
}

Teks polos dibaca langsung; toMarkdown hanya untuk PDF/DOCX/gambar. Error khusus dipetakan ke HTTP 404 di route.

7 · Ingestion

export async function ingestDoc(env: Env, key: string) {
	const text = await extract(env, key);
	const chunks = chunkText(text);
	let embedded = 0;
	let skipped = 0;

	for (let i = 0; i < chunks.length; i += 20) {
		const batch = chunks.slice(i, i + 20);
		const hashes = await Promise.all(batch.map(sha256));
		const ids = batch.map((_, j) => `${key}#${i + j}`);

		const existing = await env.DB.prepare(
			`SELECT id, hash FROM chunks WHERE id IN (${ids.map(() => "?").join(",")})`,
		).bind(...ids).all<{ id: string; hash: string }>();

		const hashById = new Map(existing.results.map((r) => [r.id, r.hash]));
		const todo: { id: string; chunk: number; text: string; hash: string }[] = [];

		batch.forEach((t, j) => {
			if (hashById.get(ids[j]) === hashes[j]) { skipped += 1; return; }
			todo.push({ id: ids[j], chunk: i + j, text: t, hash: hashes[j] });
		});

		if (!todo.length) continue;

		const vectors = await embed(env, todo.map((t) => t.text));

		await env.INDEX.upsert(
			todo.map((t, j) => ({
				id: t.id,
				values: vectors[j],
				metadata: { source: key, chunk: t.chunk },
			})),
		);

		await env.DB.batch(
			todo.map((t) =>
				env.DB.prepare(
					`INSERT OR REPLACE INTO chunks
					 (id, source, chunk_index, text, hash, created_at)
					 VALUES (?, ?, ?, ?, ?, ?)`,
				).bind(t.id, key, t.chunk, t.text, t.hash, Date.now()),
			),
		);

		embedded += todo.length;
	}

	// Rekonsiliasi: hapus chunk yang tidak lagi dihasilkan dokumen ini.
	const expected = new Set(chunks.map((_, i) => `${key}#${i}`));
	const { results: existing } = await env.DB.prepare(
		"SELECT id FROM chunks WHERE source = ?",
	).bind(key).all<{ id: string }>();

	const orphans = existing.filter((r) => !expected.has(r.id)).map((r) => r.id);

	if (orphans.length) {
		await env.DB.batch(
			orphans.map((id) => env.DB.prepare("DELETE FROM chunks WHERE id = ?").bind(id)),
		);
		await env.INDEX.deleteByIds(orphans);
	}

	return { key, chunks: chunks.length, embedded, skipped, deleted: orphans.length };
}

Batch 20, hash-skip, dan rekonsiliasi semuanya di sini. Di produksi, bungkus fungsi ini dengan Workflow: satu langkah = satu dokumen, sehingga retry tidak mengulang dokumen lain.

8 · Query

const { matches } = await env.INDEX.query(queryVector, {

9 · Ambil teks, jaga urutan

const ids = matches.map((m) => m.id);
const placeholders = ids.map(() => "?").join(",");

const { results } = await env.DB.prepare(
  `SELECT id, source, chunk_index, text
   FROM chunks WHERE id IN (${placeholders})`,
).bind(...ids).all();

const byId = new Map(results.map((r) => [r.id, r]));
const candidates = matches.flatMap((m) => {
  const row = byId.get(m.id);
  return row ? [{ ...row, score: m.score }] : [];
});

10 · Rerank

const response = await env.AI.run("@cf/baai/bge-reranker-base", {
  query: question,
  contexts: candidates.map((c) => ({ text: c.text })),
  top_k: 6,
});

const top = response
  .map((r, idx) => {
    const index = typeof r.id === "number" ? r.id : idx;
    const hit = candidates[index];
    return hit ? { ...hit, rerankScore: r.score ?? 0 } : null;
  })
  .filter((h) => h !== null)
  .sort((a, b) => b.rerankScore - a.rerankScore);

Bentuk respons: { id, score }, dengan id = indeks pada contexts. Bungkus dengan try/catch; bila gagal, pakai urutan vektor apa adanya.

11 · Fusion (hybrid)

function rrf(lists: string[][], k = 60): [string, number][] {
  const scores = new Map<string, number>();
  for (const list of lists) {
    list.forEach((id, i) => {
      scores.set(id, (scores.get(id) ?? 0) + 1 / (k + i + 1));
    });
  }
  return [...scores.entries()].sort((a, b) => b[1] - a[1]);
}

16 · Test API

import { test } from "node:test";
import assert from "node:assert/strict";

const BASE = process.env.BASE_URL ?? "http://localhost:8787";

async function countFor(source) {
	const res = await fetch(`${BASE}/health`);
	const body = await res.json();
	return body.bySource.find((r) => r.source === source)?.n ?? 0;
}

test("ingest is idempotent", async () => {
	const run = () => fetch(`${BASE}/ingest`, {
		method: "POST",
		headers: { "content-type": "application/json" },
		body: JSON.stringify({ key: "catatan/uji.md" }),
	}).then((r) => r.json());

	const before = await countFor("catatan/uji.md");
	const again = await run();
	assert.equal(again.embedded, 0);
	assert.equal(await countFor("catatan/uji.md"), before);
});

test("missing object returns 404", async () => {
	const res = await fetch(`${BASE}/ingest`, {
		method: "POST",
		headers: { "content-type": "application/json" },
		body: JSON.stringify({ key: "catatan/tidak-ada.md" }),
	});
	assert.equal(res.status, 404);
});

Seam-nya HTTP API, black-box: test tidak mengimpor modul internal. Jalankan dengan npm test saat dev server hidup. Assertion yang bergantung propagasi Vectorize memakai retry, bukan jeda tetap.
